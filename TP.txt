void update_employee(int id, char* name, char* manager) {
    Employee* temp = head;

    while (temp != NULL) {
        if (temp->id == id) {
            strcpy(temp->name, name);
            strcpy(temp->manager, manager);
            printf("Employé mis à jour avec succès.\n");
            return;
        }
        temp = temp->next;
    }
    printf("Employé avec l'ID %d introuvable.\n", id);
}
